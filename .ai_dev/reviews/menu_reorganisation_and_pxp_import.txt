BUG 2 (MEMORY LEAK): matplotlib figure not closed on error
=============================================================

SEVERITY: Medium — leaked figure objects accumulate in memory each time
a save fails; in long sessions with large trace arrays this is significant.

LOCATION: src/neuroseq/gui.py

AFFECTED METHODS:
  btn_export_traces_clicked()  lines 1084–1108
  btn_export_spikes_clicked()  lines 1143–1171

PROBLEM: In both methods the try block creates `fig` via `plt.subplots()`,
then calls `fig.savefig()`. `plt.close(fig)` is called only on the success
path (before the information message box). If `savefig()` or any preceding
line raises, the except branch runs without closing the figure.

CURRENT STRUCTURE (both methods follow this pattern):
    try:
        fig, ax = plt.subplots(...)      # fig created
        ...
        fig.savefig(filename, ...)       # can raise (e.g. disk full, bad path)
        plt.close(fig)                   # NOT reached if savefig raised
        QtWidgets.QMessageBox.information(...)
    except Exception as exc:
        self.logger.exception(...)       # fig is still open → leak
        QtWidgets.QMessageBox.critical(...)

FIX — restructure both methods to use a finally guard.
Replace the try/except block in EACH of the two methods with this pattern:

    fig = None
    try:
        fig, ax = plt.subplots(...)
        ...
        fig.savefig(filename, dpi=300, bbox_inches="tight")
        QtWidgets.QMessageBox.information(
            self,
            "Export successful",
            f"... exported to '{filename}'.",
        )
    except Exception as exc:
        self.logger.exception("Failed to export ...")
        QtWidgets.QMessageBox.critical(
            self,
            "Export error",
            f"Could not export ...: {exc}",
        )
    finally:
        if fig is not None:
            plt.close(fig)

Apply this to both btn_export_traces_clicked and btn_export_spikes_clicked.
The only change from current code is:
  1. Add `fig = None` before the try block.
  2. Remove `plt.close(fig)` from inside the try block (success path).
  3. Add a `finally:` clause after the except with `if fig is not None: plt.close(fig)`.

    def test_btn_export_traces_saves_figure(
        monkeypatch: MonkeyPatch, tmp_path: Path, qmessagebox_calls: list
    ) -> None:
        import matplotlib.pyplot as plt

        trace_data = TraceData(
            name="voltage",
            traces=np.zeros((3, 100), dtype=np.float32),
            indices=np.array([0, 1, 2], dtype=np.uint32),
            units="mV",
            dt=0.001,
        )
        output_path = tmp_path / "traces.png"
        saved_paths = []

        # monkeypatch savefig to record the call rather than write a real PNG
        original_subplots = plt.subplots
        def fake_subplots(*args, **kwargs):
            fig, ax = original_subplots(*args, **kwargs)
            original_savefig = fig.savefig
            fig.savefig = lambda path, **kw: saved_paths.append(path)
            return fig, ax

        monkeypatch.setattr(plt, "subplots", fake_subplots)
        monkeypatch.setattr(
            neuroseq.gui, "getsavefilename",
            lambda *args, **kwargs: (str(output_path), "PNG image (*.png)"),
        )

        fake_gui: Any = SimpleNamespace(
            cmb_trace=SimpleNamespace(currentText=lambda: "exp :: voltage"),
            project=SimpleNamespace(
                get_data=lambda name, dtype: trace_data,
            ),
            logger=logging.getLogger("test"),
        )
        Gui.btn_export_traces_clicked(fake_gui)

        assert len(saved_paths) == 1
        assert str(output_path) in saved_paths[0]
        assert any(c["name"] == "information" for c in qmessagebox_calls)

    def test_btn_export_traces_closes_figure_on_savefig_error(
        monkeypatch: MonkeyPatch, tmp_path: Path, qmessagebox_calls: list
    ) -> None:
        """Verifies the Bug 2 fix: figure is closed even when savefig raises."""
        import matplotlib.pyplot as plt

        trace_data = TraceData(
            name="voltage",
            traces=np.zeros((2, 50), dtype=np.float32),
            indices=np.array([0, 1], dtype=np.uint32),
            units="mV",
            dt=0.001,
        )
        output_path = tmp_path / "traces.png"
        closed_figs = []

        original_subplots = plt.subplots
        original_close = plt.close
        def fake_subplots(*args, **kwargs):
            fig, ax = original_subplots(*args, **kwargs)
            fig.savefig = lambda *a, **kw: (_ for _ in ()).throw(OSError("disk full"))
            return fig, ax
        def fake_close(fig):
            closed_figs.append(fig)
            original_close(fig)

        monkeypatch.setattr(plt, "subplots", fake_subplots)
        monkeypatch.setattr(plt, "close", fake_close)
        monkeypatch.setattr(
            neuroseq.gui, "getsavefilename",
            lambda *args, **kwargs: (str(output_path), "PNG image (*.png)"),
        )

        fake_gui: Any = SimpleNamespace(
            cmb_trace=SimpleNamespace(currentText=lambda: "exp :: voltage"),
            project=SimpleNamespace(get_data=lambda name, dtype: trace_data),
            logger=logging.getLogger("test"),
        )
        Gui.btn_export_traces_clicked(fake_gui)

        assert len(closed_figs) == 1, "figure must be closed even when savefig raises"
        assert any(c["name"] == "critical" for c in qmessagebox_calls)
