────────────────────────────────────────────────────────────────────────────
TASK 2 — PXP multi-array export
────────────────────────────────────────────────────────────────────────────

BACKGROUND
IgorPro .pxp ("Packed Experiment") is a container of typed records.
Each record = 8-byte header + payload.

    struct PackedFileRecordHeader {   // little-endian
        uint16  recordType;           // 3 = WaveRecord
        int16   version;              // 0
        int32   numDataBytes;         // length of payload in bytes
    };

Payload for a WaveRecord = full IBW bytes (including the 2-byte version
prefix).  This was verified with an igor2 round-trip: igor2.packed.load()
calls igor2.binarywave.load() on the raw payload bytes.

    RECORD_TYPE from igor2.record: {0: Unused, 1: Variables, 2: History,
    3: Wave, 4: Recreation, 5: Procedure, 6: Unused, 7: GetHistory,
    8: PackedFile, 9: FolderStart, 10: FolderEnd}

WAVE NAMING CONVENTION
Use _sanitize_igor_wave_name() (already in exporter_igor2.py) to derive the
wave name from each StimArray.name.  For inhibitory arrays, negate the values
before writing (see below).  Append a deduplication suffix ("_2", "_3", …) if
two arrays produce the same sanitized name.

INHIBITORY ARRAYS
StimType.Inhibition arrays store values as positive 0→N.  For patch clamp they
represent inhibitory current magnitude.  Export them with negated values so the
electrophysiology software sees them as negative (inhibitory) currents.
StimType.Excitation and StimType.External arrays are exported as-is.

    _PXP_WAVE_RECORD_TYPE: int = 3
    _PXP_RECORD_HEADER_FMT: str = "<HhI"   # uint16, int16, uint32 — 8 bytes

    def _write_pxp_wave_record(ibw_bytes: bytes) -> bytes:
        """Wrap full IBW bytes (including version prefix) in a PXP WaveRecord."""
        header = struct.pack(
            _PXP_RECORD_HEADER_FMT,
            _PXP_WAVE_RECORD_TYPE,
            0,
            len(ibw_bytes),
        )
        return header + ibw_bytes

Add export function:

    def export_igor2_pxp(
        output_path: str | pathlib.Path,
        stim_arrays: dict[str, "StimArray"],
        *,
        array_names: list[str] | None = None,
    ) -> Igor2PxpExport:
        """Export selected stim arrays to an Igor2 Packed Experiment (.pxp) file.

        array_names: ordered list of StimArray names to export; defaults to all.
        Inhibitory arrays (StimType.Inhibition) are negated before export.
        """
        from neuroseq.stim import StimType  # avoid circular at module level

        if array_names is None:
            array_names = list(stim_arrays.keys())

        # deduplicate wave names
        used_names: set[str] = set()
        final_wave_names: list[str] = []
        pxp_data = bytearray()

        for name in array_names:
            stim_array = stim_arrays[name]
            base = _sanitize_igor_wave_name(name)
            wave_name = base
            suffix = 2
            while wave_name in used_names:
                wave_name = f"{base}_{suffix}"[:31]
                suffix += 1
            used_names.add(wave_name)
            final_wave_names.append(wave_name)

            values = stim_array.values
            # Export mean across neurons (1-D) as the wave — one column per neuron
            # is not the IBW convention; instead we export each neuron as its own
            # wave.  See NOTE below for the chosen convention.

            # Negate inhibitory arrays
            if stim_array.stim_type == StimType.Inhibition:
                values = -values

            # Write one wave per neuron
            n_neurons = int(values.shape[1])
            for neuron_idx in range(n_neurons):
                col_values = np.asarray(values[:, neuron_idx], dtype=np.float32)
                neuron_wave_name = f"{wave_name}_n{neuron_idx}"[:31]
                # re-deduplicate neuron-suffixed names
                actual_neuron_name = neuron_wave_name
                nsfx = 2
                while actual_neuron_name in used_names:
                    actual_neuron_name = f"{wave_name}_n{neuron_idx}_{nsfx}"[:31]
                    nsfx += 1
                used_names.add(actual_neuron_name)

                ibw_bytes = _write_ibw_v5(
                    col_values,
                    wave_name=actual_neuron_name,
                    dt=float(stim_array.dt),
                    data_units=str(stim_array.units),
                    x_units="s",
                )
                pxp_data += _write_pxp_wave_record(ibw_bytes)

        path = pathlib.Path(output_path)
        path.write_bytes(bytes(pxp_data))
        return Igor2PxpExport(path=path, wave_names=final_wave_names)

NOTE ON WAVE LAYOUT:
Each stim array column (neuron) is stored as a separate 1-D wave named
<array_name>_n0, <array_name>_n1, etc.  This is the most natural mapping for
IgorPro where electrophysiology sweeps are individual 1-D waves.

    from __future__ import annotations
    from qtpy import QtWidgets
    from neuroseq.project import Project

    class Igor2PxpStimExportDialog(QtWidgets.QDialog):
        def __init__(self, project: Project, parent: QtWidgets.QWidget | None = None):
            super().__init__(parent)
            self.setWindowTitle("Export stim arrays to Igor2 PXP")
            self._project = project
            layout = QtWidgets.QVBoxLayout(self)

            layout.addWidget(QtWidgets.QLabel(
                "Select stim arrays to export.\n"
                "Inhibitory arrays will be negated (values × −1)."
            ))

            self._list = QtWidgets.QListWidget()
            for name, sa in project.stim_arrays.items():
                negate_note = " [negated]" if sa.stim_type == StimType.Inhibition else ""
                item = QtWidgets.QListWidgetItem(f"{name}{negate_note}")
                item.setData(QtCore.Qt.ItemDataRole.UserRole, name)
                item.setCheckState(QtCore.Qt.CheckState.Checked)
                self._list.addItem(item)
            layout.addWidget(self._list)

            btns = QtWidgets.QDialogButtonBox(
                QtWidgets.QDialogButtonBox.StandardButton.Ok
                | QtWidgets.QDialogButtonBox.StandardButton.Cancel
            )
            btns.accepted.connect(self.accept)
            btns.rejected.connect(self.reject)
            layout.addWidget(btns)

        @property
        def selected_array_names(self) -> list[str]:
            result = []
            for i in range(self._list.count()):
                item = self._list.item(i)
                if item.checkState() == QtCore.Qt.CheckState.Checked:
                    result.append(item.data(QtCore.Qt.ItemDataRole.UserRole))
            return result

    def test_pxp_roundtrip_single_array(tmp_path):
        """PXP with one excitatory array round-trips correctly."""
        import igor2.packed as pkg
        stim = StimArray(values=np.array([[1.0, 2.0], [3.0, 4.0]]), dt=0.001,
                         units="pA", stim_type=StimType.Excitation, name="Exc")
        out = tmp_path / "test.pxp"
        report = export_igor2_pxp(out, {"Exc": stim})
        assert out.exists()
        records, _ = pkg.load(out)
        wave_names = [list(r.wave['wave']['wave_header']['bname']) for r in records
                      if hasattr(r, 'wave')]
        # two waves: Exc_n0, Exc_n1
        assert len(records) == 2

    def test_pxp_inhibitory_negated(tmp_path):
        """Inhibitory stim array values are negated in the exported waves."""
        import igor2.packed as pkg
        vals = np.array([[1.0], [2.0], [3.0]])
        stim = StimArray(values=vals, dt=0.001, units="1",
                         stim_type=StimType.Inhibition, name="Inh")
        out = tmp_path / "inh.pxp"
        export_igor2_pxp(out, {"Inh": stim})
        records, _ = pkg.load(out)
        wave_data = records[0].wave['wave']['wData']
        assert np.allclose(wave_data, [-1.0, -2.0, -3.0])

    def test_pxp_multi_array(tmp_path):
        """PXP with multiple arrays writes one wave per neuron per array."""
        import igor2.packed as pkg
        exc = StimArray(values=np.ones((10, 3)), dt=0.001, units="pA",
                        stim_type=StimType.Excitation, name="Exc")
        inh = StimArray(values=np.ones((10, 2)), dt=0.001, units="1",
                        stim_type=StimType.Inhibition, name="Inh")
        out = tmp_path / "multi.pxp"
        export_igor2_pxp(out, {"Exc": exc, "Inh": inh})
        records, _ = pkg.load(out)
        assert len(records) == 5  # 3 + 2

    def test_pxp_external_not_negated(tmp_path):
        """External stim arrays are exported as-is (not negated)."""
        import igor2.packed as pkg
        vals = np.array([[-1.0], [2.0]])
        stim = StimArray(values=vals, dt=0.001, units="pA",
                         stim_type=StimType.External, name="Ext")
        out = tmp_path / "ext.pxp"
        export_igor2_pxp(out, {"Ext": stim})
        records, _ = pkg.load(out)
        wave_data = records[0].wave['wave']['wData']
        assert np.allclose(wave_data, [-1.0, 2.0])
