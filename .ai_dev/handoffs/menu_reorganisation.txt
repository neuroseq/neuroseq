    # ── Import stimulus wrappers ──────────────────────────────────────────────

    def btn_import_stimulus_light_clicked(self) -> None:
        """Pre-dialog: choose software, then import a light stimulus waveform."""
        software, ok = QtWidgets.QInputDialog.getItem(
            self,
            "Import light stimulus",
            "Import from:",
            ["SysCon2"],
            0,
            False,
        )
        if not ok:
            return
        if software == "SysCon2":
            self.btn_import_syscon2_seq_clicked()

            ["Igor2"],

    def btn_import_trace_patchclamp_clicked(self) -> None:

        Convention:
          Current Clamp (CC) → recorded signal is voltage (mV)

        determined by the units stored in the imported TraceData.
        """
        software, ok = QtWidgets.QInputDialog.getItem(
            self,
            "Import patch clamp traces",
            "Import from:",
            ["Igor2"],
            0,
            False,
        )
        if not ok:
            return
        if software == "Igor2":
            self.btn_import_patchclamp_data_clicked()

    def btn_export_traces_clicked(self) -> None:
        """Export the currently displayed trace data as a publication-quality figure.

        Exports all neuron traces for the selected TraceData as a single matplotlib
        figure. The Y-axis label uses the units stored in the TraceData (e.g. 'mV'
        for current-clamp recordings, 'pA' for voltage-clamp recordings). DPI is 300.
        """
        import matplotlib.pyplot as plt

        trace_name = self.cmb_trace.currentText()
        if not trace_name:
            QtWidgets.QMessageBox.information(
                self,
                "No trace data",
                "No trace data is currently selected in the Traces panel.",
            )
            return
        trace_data = self.project.get_data(trace_name, TraceData)
        if trace_data is None:
            QtWidgets.QMessageBox.information(
                self,
                "No trace data",
                f"Could not retrieve trace data '{trace_name}'.",
            )
            return

        filename, _ = getsavefilename(
            self,
            "Export traces",
            f"{trace_name.replace(' :: ', '_')}_traces.png",
            "PNG image (*.png);;SVG vector (*.svg);;PDF document (*.pdf)",
        )
        if not filename:
            return

        try:
            n_neurons, n_timepoints = trace_data.traces.shape
            times = np.arange(n_timepoints) * trace_data.dt
            fig, ax = plt.subplots(figsize=(12, max(4, n_neurons * 0.5 + 2)))
            for i in range(n_neurons):
                ax.plot(times, trace_data.traces[i], linewidth=0.8, alpha=0.85)
            ax.set_xlabel("Time (s)", fontsize=12)
            ax.set_ylabel(trace_data.units or "Amplitude", fontsize=12)
            ax.set_title(trace_name, fontsize=13)
            ax.tick_params(labelsize=10)
            fig.tight_layout()
            fig.savefig(filename, dpi=300, bbox_inches="tight")
            plt.close(fig)
            QtWidgets.QMessageBox.information(
                self,
                "Export successful",
                f"Traces exported to '{filename}'.",
            )
        except Exception as exc:
            self.logger.exception("Failed to export traces")
            QtWidgets.QMessageBox.critical(
                self,
                "Export error",
                f"Could not export traces: {exc}",
            )

    def btn_export_spikes_clicked(self) -> None:
        """Export the currently displayed spike data as a publication-quality raster.

        Produces a scatter-plot raster (spike time vs neuron index) saved at 300 DPI.
        """
        import matplotlib.pyplot as plt

        spike_name = self.cmb_spike.currentText()
        if not spike_name:
            QtWidgets.QMessageBox.information(
                self,
                "No spike data",
                "No spike data is currently selected in the Spikes panel.",
            )
            return
        spike_data = self.project.get_data(spike_name, SpikeData)
        if spike_data is None:
            QtWidgets.QMessageBox.information(
                self,
                "No spike data",
                f"Could not retrieve spike data '{spike_name}'.",
            )
            return

        filename, _ = getsavefilename(
            self,
            "Export spikes",
            f"{spike_name.replace(' :: ', '_')}_spikes.png",
            "PNG image (*.png);;SVG vector (*.svg);;PDF document (*.pdf)",
        )
        if not filename:
            return

        try:
            fig, ax = plt.subplots(figsize=(12, 5))
            ax.scatter(
                spike_data.spike_times,
                spike_data.spike_indices,
                s=1,
                c="black",
                alpha=0.7,
                linewidths=0,
            )
            ax.set_xlabel("Time (s)", fontsize=12)
            ax.set_ylabel("Neuron index", fontsize=12)
            ax.set_title(spike_name, fontsize=13)
            ax.tick_params(labelsize=10)
            fig.tight_layout()
            fig.savefig(filename, dpi=300, bbox_inches="tight")
            plt.close(fig)
            QtWidgets.QMessageBox.information(
                self,
                "Export successful",
                f"Spikes exported to '{filename}'.",
            )
        except Exception as exc:
            self.logger.exception("Failed to export spikes")
            QtWidgets.QMessageBox.critical(
                self,
                "Export error",
                f"Could not export spikes: {exc}",
            )

IMPLEMENTATION NOTES
=============================================================

1. TOOLBAR: add_action() accepts both menu and toolbar. For all items inside
   import_data_menu and export_data_menu, pass toolbar=None. Only the top-level
   actions (New, Open, Import from project, Save, Edit model, Generate Synthetic Data)
   keep their toolbar presence.

2. SHORTCUT CONFLICT: "Ctrl+Shift+E" was previously assigned to both "&Export data"
   (File menu) AND "&Connect brian2 external stimulation function" (Experiment menu).
   In the new structure, Ctrl+Shift+E moves to "Export simulation results" in the
   export sub-menu. The Experiment menu item still has its own Ctrl+Shift+E shortcut —
   check whether Qt produces a conflict warning; if so, remove the shortcut from the
   Experiment action (it's less commonly used).

3. MATPLOTLIB IMPORT: import matplotlib.pyplot as plt is done lazily inside the
   two export methods, not at module level. matplotlib is available (Brian2 dependency).
   plt.close(fig) must always be called to avoid memory leaks from held figure objects.

4. TRACE DATA SHAPE: TraceData.traces is shape (n_neurons, n_timepoints).
   SpikeData.spike_times and spike_indices are 1-D arrays.
   project.get_data(name, DataType) takes the name exactly as it appears in cmb_trace /
   cmb_spike (format: "ExperimentName :: DataName" or "Simulation :: ...").

5. FIGURE HEIGHT FOR TRACES: use max(4, n_neurons * 0.5 + 2) to scale the figure
   vertically with neuron count, capped at a minimum of 4 inches. Avoid figures that
   are unreadably tiny for small datasets or enormous for large ones.

6. NO NEW FILES: this entire change lives in gui.py (and two lines in test_gui.py).
   No new .py files, dialogs, or modules are needed.

7. MYPY / RUFF:
   - All new methods need return type annotation -> None.
   - matplotlib import inside function body avoids a missing-stub warning.
   - isort force-single-line is enforced; no new top-level imports are added.
