PROBLEM
-------
Today `File → Import data → Import trace → Import patch clamp` asks the
user to choose "Igor2" but then opens `PatchClampImportDialog`, which
only reads `.npy` files. The CC/VC convention note we just added to the
pre-dialog reinforces the expectation that Igor .ibw/.pxp/.csv should
work here the same way they do for stimuli — and they don't.

The stimulus side (`File → Import data → Import stimulus → Import patch
clamp`) already has a fully-featured Igor2 dialog — `IgorImportDialog` —
with `.ibw`/`.pxp`/`.csv` loading, tolerant packed-experiment scanning,
per-wave picker, dt adjustment, and multi-wave "stay open until Close"
behaviour. We want the trace side to behave the same way, routed to
`TraceData` / `Experiment` instead of `StimArray`.

    def import_igor_wave_into_trace_data(
        trace_data: TraceData,
        prepared_wave: PreparedIgorWave,
        *,
        neuron_id: int,
        replace_existing: bool = True,
        extend_to_fit: bool = True,
    ) -> IgorTraceImportReport:
        """Append or replace a single neuron row in an existing TraceData.

        - If ``neuron_id`` is already in ``trace_data.indices`` and
          ``replace_existing`` is True, that row is overwritten with the
          imported wave (with length handling per ``extend_to_fit``).
        - If ``neuron_id`` is already present and ``replace_existing`` is
          False, raises ValueError.
        - If ``neuron_id`` is new, appends a new row. Existing rows and
          the new row are padded to a common length (either the existing
          length or the imported length, depending on ``extend_to_fit``).

        Raises ValueError if the imported target_dt does not match
        ``trace_data.dt`` (caller is expected to remap first — mirror the
        stim behaviour where `update_stim_array_properties` is used
        before calling `import_igor_wave_into_stim_array`).
        """
        if abs(float(trace_data.dt) - float(prepared_wave.target_dt)) > _DT_TOLERANCE_SECONDS:
            raise ValueError(
                "Target TraceData dt must match the imported wave target dt "
                f"({float(trace_data.dt):.9f}s != {float(prepared_wave.target_dt):.9f}s)"
            )

        imported_values = prepared_wave.values.astype(np.float32, copy=False)
        n_rows_existing, n_time_existing = trace_data.traces.shape
        n_time_imported = int(imported_values.size)

        indices_list = [int(x) for x in trace_data.indices.tolist()]
        row_index = indices_list.index(int(neuron_id)) if int(neuron_id) in indices_list else -1

        if row_index >= 0 and not replace_existing:
            raise ValueError(
                f"Neuron ID {neuron_id} already exists in trace data '{trace_data.name}'; "
                "enable 'Replace existing' to overwrite it."
            )

        if extend_to_fit:
            n_time_target = max(n_time_existing, n_time_imported)
        else:
            n_time_target = n_time_existing

        if n_time_target != n_time_existing:
            padded = np.zeros((n_rows_existing, n_time_target), dtype=trace_data.traces.dtype)
            padded[:, :n_time_existing] = trace_data.traces
            trace_data.traces = padded

        truncated = max(0, n_time_imported - n_time_target)
        written_len = min(n_time_imported, n_time_target)

        new_row = np.zeros((1, n_time_target), dtype=trace_data.traces.dtype)
        new_row[0, :written_len] = imported_values[:written_len]

        if row_index >= 0:
            trace_data.traces[row_index] = new_row[0]
        else:
            trace_data.traces = np.concatenate([trace_data.traces, new_row], axis=0)
            trace_data.indices = np.concatenate(
                [trace_data.indices, np.array([int(neuron_id)], dtype=np.uint32)]
            )

        logger.info(
            "Imported Igor wave '%s' into TraceData '%s' neuron %d: %d -> %d samples",
            prepared_wave.name,
            trace_data.name,
            int(neuron_id),
            prepared_wave.original_sample_count,
            written_len,
        )
        return IgorTraceImportReport(
            target_experiment_name="",  # filled in by caller which knows the experiment
            target_trace_data_name=trace_data.name,
            target_neuron_id=int(neuron_id),
            source_samples=prepared_wave.original_sample_count,
            imported_samples=written_len,
            source_dt=float(prepared_wave.source_dt),
            target_dt=float(prepared_wave.target_dt),
            resampled=prepared_wave.resampled,
            truncated_samples=truncated,
        )

    def add_data_to_experiment(
        self,
        experiment_name: str,
        data: TraceData | SpikeData | SynapseData,
    ) -> None:
        """Append a data entry to an existing experiment and keep
        duration/timepoints coherent. If the experiment does not exist,
        raises KeyError. If a TraceData extends beyond the current
        experiment duration, bump duration and append the new timepoint.
        """
        experiment = self.experiments.get(experiment_name)
        if experiment is None:
            raise KeyError(f"Experiment '{experiment_name}' does not exist")
        experiment.data.append(data)
        if isinstance(data, TraceData):
            new_duration = float(data.traces.shape[1] * data.dt)
            if new_duration > experiment.duration:
                experiment.duration = new_duration
                if new_duration not in experiment.timepoints:
                    experiment.timepoints.append(new_duration)
        self.update_limits()

3h) Helper methods to implement on the class

    def _resolve_new_target_experiment_name(self) -> str:
        current = self.new_experiment_combo.currentText()
        if current == "(New experiment)":
            name = self.new_experiment_name_edit.text().strip()
            if not name:
                raise ValueError("Experiment name cannot be empty")
            if name in self.project.experiments:
                raise ValueError(f"Experiment '{name}' already exists")
            return name
        return current

    def _commit_new_trace_data(
        self, experiment_name: str, trace_data: TraceData
    ) -> None:
        if experiment_name in self.project.experiments:
            # Disallow duplicate TraceData name within the experiment.
            existing_names = [
                getattr(d, "name", "")
                for d in self.project.experiments[experiment_name].data
            ]
            if trace_data.name in existing_names:
                raise ValueError(
                    f"Trace data '{trace_data.name}' already exists "
                    f"in experiment '{experiment_name}'"
                )
            self.project.add_data_to_experiment(experiment_name, trace_data)
        else:
            self.project.add_experimental_data(experiment_name, [trace_data])

    def _resolve_existing_trace_data(
        self, experiment_name: str, trace_name: str
    ) -> TraceData | None:
        experiment = self.project.experiments.get(experiment_name)
        if experiment is None:
            return None
        for data in experiment.data:
            if isinstance(data, TraceData) and data.name == trace_name:
                return data
        return None

    def _confirm_and_remap_trace_dt(
        self, target_trace: TraceData, experiment_name: str, new_dt: float
    ) -> bool:
        """Mirror the stim dialog's dt-remap confirm. Because TraceData
        has no project-level `update_*_properties` helper, do the remap
        in place: resample existing rows with zero-order hold to the new
        dt. Users who dislike the change can click No to abort."""
        msg = (
            "Changing the stimulation timestep will remap trace data "
            f"'{target_trace.name}' from "
            f"{format_milliseconds(float(target_trace.dt) * 1000.0)} ms "
            f"to {format_milliseconds(new_dt * 1000.0)} ms. "
            "Trace timing may shift by up to one sample. Continue?"
        )
        answer = QtWidgets.QMessageBox.question(
            self, "Change trace data timestep", msg,
            QtWidgets.QMessageBox.StandardButton.Yes
            | QtWidgets.QMessageBox.StandardButton.No,
        )
        if answer != QtWidgets.QMessageBox.StandardButton.Yes:
            return False
        # Zero-order-hold remap of every existing row.
        from neuroseq.importer_igor2 import resample_wave_zero_order_hold
        new_rows = [
            resample_wave_zero_order_hold(
                target_trace.traces[i],
                source_dt=float(target_trace.dt),
                target_dt=float(new_dt),
            )
            for i in range(target_trace.traces.shape[0])
        ]
        target_trace.traces = np.stack(new_rows, axis=0).astype(np.float32, copy=False)
        target_trace.dt = float(new_dt)
        return True
