def is_pxp_file(path: str | Path) -> bool:
    """Return True if *path* has a .pxp extension (case-insensitive)."""
    return Path(path).suffix.lower() == ".pxp"


def load_igor_pxp_file(path: str | Path) -> dict[str, IgorWave]:
    """Load all 1-D waves from an Igor Packed Experiment (.pxp) file.

    Returns a dict mapping slash-joined folder paths to IgorWave objects.
    Example key: "root/stims/HodgkinHuxley" or just "HodgkinHuxley" if at root.

    Only 1-D waves are included; multi-dimensional waves are skipped with a
    debug log message. Raises ImportError if igor2 is not installed. Raises
    ValueError if the file contains no importable 1-D waves.
    """
    path = Path(path)
    try:
        packed = importlib.import_module("igor2.packed")
        wave_record_module = importlib.import_module("igor2.record.wave")
    except ImportError as exc:
        raise ImportError(
            "Reading Igor Packed Experiment files requires the optional 'igor2' dependency."
        ) from exc

    WaveRecord = wave_record_module.WaveRecord

    _records, filesystem = packed.load(path)

    waves: dict[str, IgorWave] = {}

    def _collect(dirpath: list[object], key: object, value: object) -> None:
        if not isinstance(value, WaveRecord):
            return
        # Build a human-readable path key
        parts = [
            seg.decode("utf-8", errors="ignore") if isinstance(seg, (bytes, bytearray)) else str(seg)
            for seg in dirpath
        ]
        wave_name_str = (
            key.decode("utf-8", errors="ignore") if isinstance(key, (bytes, bytearray)) else str(key)
        )
        parts.append(wave_name_str)
        path_key = "/".join(parts)

        try:
            igor_wave = parse_igor_binary_wave_data(
                value.wave,
                fallback_name=wave_name_str,
            )
        except ValueError as exc:
            logger.debug("Skipping wave %s: %s", path_key, exc)
            return

        waves[path_key] = igor_wave

    packed.walk(filesystem, _collect)

    if not waves:
        raise ValueError(
            f"No importable 1-D waves found in packed experiment file '{path.name}'"
        )

    return waves

        else:
            self.loaded_wave = None
            self._update_preview()
            self._update_import_button_state()

def _make_fake_packed_module(filesystem: dict) -> types.ModuleType:
    """Build a fake igor2.packed module for monkeypatching."""
    import types as _types

    mod = _types.ModuleType("igor2.packed")

    def load(path):
        return [], filesystem

    def walk(fs, callback, dirpath=None):
        if dirpath is None:
            dirpath = []
        for key, value in sorted(fs.items()):
            callback(dirpath, key, value)
            if isinstance(value, dict):
                walk(fs=value, callback=callback, dirpath=dirpath + [key])

    mod.load = load
    mod.walk = walk
    return mod


def _make_fake_wave_record(wave_dict: dict) -> object:
    """Build a fake WaveRecord for testing."""
    class FakeWaveRecord:
        wave = wave_dict
    return FakeWaveRecord()


def test_load_igor_pxp_file_returns_all_1d_waves(
    monkeypatch: pytest.MonkeyPatch,
    tmp_path: Path,
) -> None:
    pxp_path = tmp_path / "test.pxp"
    pxp_path.write_bytes(b"fake pxp")

    wave_dict = {
        "version": 5,
        "wave": {
            "wave_header": {
                "bname": b"MyWave\x00",
                "sfA": [0.001, 0.0, 0.0, 0.0],
                "dataUnits": b"pA\x00",
                "dimUnits": [b"s\x00", b"", b"", b""],
            },
            "wData": np.array([1.0, 2.0, 3.0], dtype=np.float32),
            "note": b"",
        },

    fake_packed = _make_fake_packed_module(
        {b"root": {b"MyWave": _make_fake_wave_record(wave_dict)}}
    )

    class FakeWaveRecordModule:
        WaveRecord = type(_make_fake_wave_record(wave_dict))

    original_import = importlib.import_module

    def fake_import(name, package=None):
        if name == "igor2.packed":
            return fake_packed
        if name == "igor2.record.wave":
            return FakeWaveRecordModule()
        return original_import(name, package)

    monkeypatch.setattr(importlib, "import_module", fake_import)

    waves = load_igor_pxp_file(pxp_path)

    assert len(waves) == 1
    key = next(iter(waves))
    assert "MyWave" in key
    wave = waves[key]
    assert wave.name == "MyWave"
    assert wave.source_dt == pytest.approx(0.001)
    assert wave.units == "pA"
    np.testing.assert_allclose(wave.values, np.array([1.0, 2.0, 3.0]))
